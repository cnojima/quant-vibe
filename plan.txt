  1.3 API Client Setup

  // src/api/client.ts
  import axios from 'axios';

  const apiClient = axios.create({
    baseURL: import.meta.env.VITE_API_URL ||
  'http://localhost:8000/api',
    timeout: 30000,
  });

  // Add JWT token to requests
  apiClient.interceptors.request.use((config) => {
    const token = localStorage.getItem('auth_token');
    if (token) {
      config.headers.Authorization = `Bearer ${token}`;
    }
    return config;
  });

  // Handle 401 errors (token expired)
  apiClient.interceptors.response.use(
    (response) => response,
    (error) => {
      if (error.response?.status === 401) {
        localStorage.removeItem('auth_token');
        window.location.href = '/login';
      }
      return Promise.reject(error);
    }
  );

  export default apiClient;

  1.4 WebSocket Hook

  // src/hooks/useWebSocket.ts
  import { useEffect, useRef, useState } from 'react';

  export function useWebSocket<T>(url: string) {
    const [data, setData] = useState<T | null>(null);
    const [isConnected, setIsConnected] = useState(false);
    const ws = useRef<WebSocket | null>(null);

    useEffect(() => {
      const token = localStorage.getItem('auth_token');
      const wsUrl = `${url}?token=${token}`;

      ws.current = new WebSocket(wsUrl);

      ws.current.onopen = () => setIsConnected(true);
      ws.current.onclose = () => setIsConnected(false);
      ws.current.onmessage = (event) => {
        const message = JSON.parse(event.data);
        setData(message);
      };

      return () => ws.current?.close();
    }, [url]);

    return { data, isConnected };
  }

  // src/pages/Login.tsx
  import { useState } from 'react';
  import { useNavigate } from 'react-router-dom';
  import apiClient from '../api/client';

  export function Login() {
    const [username, setUsername] = useState('');
    const [password, setPassword] = useState('');
    const [error, setError] = useState('');
    const navigate = useNavigate();

    const handleLogin = async (e: React.FormEvent) => {
      e.preventDefault();
      try {
        const formData = new FormData();
        formData.append('username', username);
        formData.append('password', password);

        const response = await apiClient.post('/auth/login',
  formData);
        localStorage.setItem('auth_token',
  response.data.access_token);
        navigate('/');
      } catch (err) {
        setError('Invalid credentials');
      }
    };

    return (
      <div className="min-h-screen flex items-center 
  justify-center">
        <form onSubmit={handleLogin} className="max-w-md w-full">
          <h1 className="text-2xl font-bold mb-4">QuantVibe
  Admin</h1>
          {error && <div className="text-red-500 
  mb-2">{error}</div>}
          <input
            type="text"
            placeholder="Username"
            value={username}
            onChange={(e) => setUsername(e.target.value)}
            className="w-full p-2 border rounded mb-2"

            type="password"

  2.3 Token Manager

  // src/pages/TokenManager.tsx
  import { useQuery, useMutation } from '@tanstack/react-query';
  import apiClient from '../api/client';

  interface TokenStatus {
    access_token_valid: boolean;
    access_token_expiry: string;
    refresh_token_valid: boolean;
    refresh_token_expiry: string;
  }

  export function TokenManager() {
    const { data: status } = useQuery({
      queryKey: ['token-status'],
      queryFn: async () => {
        const response = await
  apiClient.get<TokenStatus>('/tokens/status');
        return response.data;
      },
      refetchInterval: 60000, // Refresh every minute
    });

    const refreshMutation = useMutation({
      mutationFn: async () => {
        await apiClient.post('/tokens/refresh');
      },
    });

    const getTimeRemaining = (expiry: string) => {
      const diff = new Date(expiry).getTime() - Date.now();
      const minutes = Math.floor(diff / 60000);
      const seconds = Math.floor((diff % 60000) / 1000);
      return `${minutes}m ${seconds}s`;
    };

    return (
      <div className="p-6">
        <h1 className="text-3xl font-bold mb-6">Schwab API
  Token</h1>

        <div className="bg-white rounded-lg shadow p-6">
          <div className="space-y-4">
            <div>
              <h3 className="font-semibold">Access Token</h3>
              <p className={status?.access_token_valid ? 
  'text-green-600' : 'text-red-600'}>
                {status?.access_token_valid ? '✓ Valid' : '✗
  Expired'}
              </p>
              {status?.access_token_expiry && (
                <p className="text-sm text-gray-600">
                  Expires in:
  {getTimeRemaining(status.access_token_expiry)}
                </p>
              )}
            </div>

            <div>
              <h3 className="font-semibold">Refresh Token</h3>
              <p className={status?.refresh_token_valid ? 
  'text-green-600' : 'text-red-600'}>
                {status?.refresh_token_valid ? '✓ Valid' : '✗
  Expired'}
              </p>
              {status?.refresh_token_expiry && (
                <p className="text-sm text-gray-600">
                  Expires in:
  {getTimeRemaining(status.refresh_token_expiry)}
                </p>
              )}
            </div>

            <div className="flex gap-2">
              <button
                onClick={() => refreshMutation.mutate()}
                disabled={refreshMutation.isPending}
                className="bg-blue-500 text-white px-4 py-2
  rounded"
              >
                {refreshMutation.isPending ? 'Refreshing...' :
  'Refresh Token'}
              </button>
            </div>
          </div>
        </div>
      </div>
    );
  }

  export function Backtest() {
    const [selectedStrategy, setSelectedStrategy] = useState('');
    const [params, setParams] = useState<Record<string, any>>({});
    const [backtestId, setBacktestId] = useState<string |
  null>(null);

    const { data: strategies } = useQuery({
      queryKey: ['strategies'],
      queryFn: async () => {
        const response = await
  apiClient.get('/backtests/strategies');
        return response.data;
      },
    });

    const runMutation = useMutation({
      mutationFn: async (payload: any) => {
        const response = await apiClient.post('/backtests/run',
  payload);
        return response.data;
      },
      onSuccess: (data) => {
        setBacktestId(data.backtest_id);
      },
    });

    const { data: results } = useQuery({
      queryKey: ['backtest-results', backtestId],
      queryFn: async () => {
        const response = await
  apiClient.get(`/backtests/${backtestId}/results`);
        return response.data;
      },
      enabled: !!backtestId,
    });

    const handleRun = () => {
      runMutation.mutate({
        strategy_name: selectedStrategy,
        params: params,
        start_date: '2025-12-01',
        end_date: '2025-12-25',
      });
    };

    return (
      <div className="p-6">
        <h1 className="text-3xl font-bold mb-6">Backtest
  Runner</h1>

        {/* Strategy Selector */}
        <div className="bg-white rounded-lg shadow p-6 mb-6">
          <label className="block mb-2 
  font-semibold">Strategy</label>
          <select
            value={selectedStrategy}
            onChange={(e) => setSelectedStrategy(e.target.value)}
            className="w-full p-2 border rounded"
          >
            <option value="">Select a strategy...</option>
            {strategies?.map((s: any) => (
              <option key={s.name} value={s.name}>{s.name}</option>
            ))}
          </select>

          {/* Parameter Form (dynamic based on strategy) */}
          {selectedStrategy && (
            <div className="mt-4 space-y-2">
              <input
                type="number"
                placeholder="Spread Width"
                onChange={(e) => setParams({...params,
  spread_width: parseFloat(e.target.value)})}
                className="w-full p-2 border rounded"
              />
              {/* Add more parameter inputs based on strategy */}
            </div>
          )}

          <button
            onClick={handleRun}
            disabled={!selectedStrategy || runMutation.isPending}
            className="mt-4 bg-blue-500 text-white px-4 py-2 
  rounded"
          >
            {runMutation.isPending ? 'Running...' : 'Run Backtest'}
          </button>
        </div>

        {/* Results */}
        {results && (
          <div className="bg-white rounded-lg shadow p-6">
            <h2 className="text-xl font-semibold mb-4">Results</h2>
            <div className="grid grid-cols-3 gap-4 mb-6">
              <div>
                <p className="text-sm text-gray-600">Total
  Trades</p>
                <p className="text-2xl 
  font-bold">{results.total_trades}</p>

                  ${results.total_pnl.toFixed(2)}

      """Get YAML configuration"""
